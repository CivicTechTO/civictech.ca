{{- /* Returns the processable resource for a meetup image, or nil. A few archive images carry a .png name over WebP or AVIF bytes; libvips sniffs them, but Hugo goes by extension, so those are re-typed by magic bytes. Only .png files are read, so the other images are never loaded here. */ -}} {{- $img := resources.Get (printf "images/events/%s" .) }} {{- if and $img (eq (path.Ext $img.Name) ".png") }} {{- $head := substr $img.Content 0 12 }} {{- $ext := "" }} {{- if hasPrefix $head "RIFF" }}{{ $ext = "webp" }}{{ else if eq (substr $head 4 8) "ftypavif" }}{{ $ext = "avif" }}{{ end }} {{- with $ext }} {{- $img = resources.FromString (printf "images/events/%s.%s" (path.BaseName $img.Name) .) $img.Content }} {{- end }} {{- end }} {{- return $img }}
