{{- /* Returns the processable resource for a meetup image, or nil. ARCHIVE-COMPAT(mislabeled-images): hacknight_528–530.png are WebP and hacknight_549, 551, 552.png are AVIF. libvips sniffs content, but Hugo goes by extension, so .png files are re-typed by magic bytes. Only .png files are read, so other images are never loaded here. A mislabelled file with another extension fails the build at Resize. Remove once the archive stores these with correct extensions. */ -}} {{- $img := resources.Get (printf "images/events/%s" .) }} {{- if and $img (eq (path.Ext $img.Name) ".png") }} {{- $head := substr $img.Content 0 12 }} {{- $ext := "" }} {{- if hasPrefix $head "RIFF" }}{{ $ext = "webp" }}{{ else if eq (substr $head 4 8) "ftypavif" }}{{ $ext = "avif" }}{{ end }} {{- with $ext }} {{- $img = resources.FromString (printf "images/events/%s.%s" (path.BaseName $img.Name) .) $img.Content }} {{- end }} {{- end }} {{- return $img }}
