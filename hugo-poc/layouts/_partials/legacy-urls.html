{{- /* Old Jekyll URL → Hugo URL for every page whose URL changed. Jekyll used the file name verbatim (minus trailing dots), so "/people/Lilly Tong/"; Hugo sanitizes it to "/people/Lilly-Tong/". Paths are returned decoded. */ -}} {{- $redirects := slice }} {{- range slice "people" "venues" "projects" "organizations" "announcements" "resources" }} {{- range where site.RegularPages "Section" . }} {{- $old := printf "/%s/%s/" .Section (replaceRE `\.+$` "" .File.ContentBaseName) }} {{- if ne $old .RelPermalink }} {{- $redirects = $redirects | append (dict "from" $old "to" .RelPermalink) }} {{- end }} {{- end }} {{- end }} {{- return $redirects }}
