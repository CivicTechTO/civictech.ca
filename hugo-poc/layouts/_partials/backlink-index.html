{{- /* Builds a wikilink → pages map once per (sections, fields) pair. Called via partialCached, so the whole site is scanned once rather than once per page as backlinks.html does in Jekyll. Pages keep collection order: meetups by date, everything else by file path (byte order, as Jekyll). */ -}} {{- $index := newScratch }} {{- range .sections }} {{- $pages := partial "sort-by-path.html" (where site.RegularPages "Section" .) }} {{- if eq . "events" }}{{ $pages = sort $pages "Date" }}{{ end }} {{- range $page := $pages }} {{- $targets := slice }} {{- range $.fields }} {{- with index $page.Params . }} {{- range (cond (reflect.IsSlice .) . (slice .)) }} {{- $targets = $targets | append (findRE `\[\[[^\]]+\]\]` (printf "%v" .)) }} {{- end }} {{- end }} {{- end }} {{- range uniq $targets }}{{ $index.Add . (slice $page) }}{{ end }} {{- end }} {{- end }} {{- return $index }}
